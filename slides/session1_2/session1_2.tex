\begin{frame}{From Session 1.1 to Session 1.2}
\begin{itemize}
\item Session 1.1 established a static ERGM as a probability law on an explicitly declared network support.
\item Here, we ask whether a \term{structural specification} is computationally stable and scientifically adequate after it is fit.
\item An \term{edges-only} or small baseline remains useful: it makes every later refinement a comparison against an explicit reference.
\item A more elaborate formula is \emph{not} automatically better: it can be redundant, unidentified, poorly mixed, or concentrated on unrealistic graphs.
\item \alert{Rule:} model fitting, numerical convergence, and generative adequacy are separate claims that require separate evidence.
\end{itemize}
\end{frame}

\begin{frame}{Three questions after an ERGM is specified}
\begin{enumerate}
\item \term{Does the sampler explore?} Inspect MCMC traces, autocorrelation, and sampled-statistic distributions.
\item \term{Has the estimation target stabilized?} Examine estimating-equation or projected-score evidence, Monte Carlo uncertainty, warnings, and repeated fits.
\item \term{Does the fitted distribution reproduce relevant structure?} Compare observed diagnostics with networks simulated from the fitted model.
\end{enumerate}
\begin{block}{What is insufficient?}
An optimizer stopping, small printed standard errors, or a coefficient with a small nominal $p$-value does not by itself answer any of the three questions.
\end{block}
\end{frame}

\begin{frame}{Curved exponential families}
\begin{itemize}
\item A \term{curved ERGM} has a natural parameter that is a nonlinear function of a lower-dimensional parameter: $\eta(\theta)\in\R^q$ with $q>p=\dim(\theta)$.
\item Its probability law is
\[
\Pr_{\theta}(Y=y)=\frac{\exp\{\eta(\theta)^{\mathsf T}g(y)\}}{\kappa\{\eta(\theta)\}}.
\]
\item \term{Curvature} restricts the natural-parameter vector to a lower-dimensional manifold rather than allowing each configuration count an unrelated coefficient.
\item The aim is parsimony: represent a family of related local configurations while controlling the influence of very high counts.
\end{itemize}
\end{frame}

\begin{frame}{Curved score and information}
Let $D_{\eta}(\theta)=\partial\eta(\theta)/\partial\theta^{\mathsf T}$ be the $q\times p$ Jacobian.
\begin{itemize}
\item The \term{score} is
\[
U(\theta)=D_{\eta}(\theta)^{\mathsf T}\left[g(y)-\E_{\theta}\{g(Y)\}\right].
\]
\item The expected Fisher information is
\[
I(\theta)=D_{\eta}(\theta)^{\mathsf T}\Var_{\theta}\{g(Y)\}D_{\eta}(\theta).
\]
\item At a curved-family maximum-likelihood estimate, the \term{projected score} is near zero relative to Monte Carlo uncertainty.
\item Componentwise equality of every underlying configuration count is \emph{not} the curved-model convergence criterion.
\end{itemize}
\end{frame}

\begin{frame}{Canonical and mean-value parameterizations}
\begin{itemize}
\item The \term{canonical parameter} describes a graph distribution through coefficients in the exponential family.
\item The \term{mean-value parameter} is $\mu(\theta)=\E_{\theta}\{g(Y)\}$: the vector of expected sufficient statistics.
\item For a regular minimal canonical family with finite support, interior mean values correspond one-to-one with canonical parameters.
\item Curved models restrict attainable mean values to the image of a lower-dimensional parameterization; full-family geometric intuitions must therefore not be transferred mechanically.
\item Near a boundary of the convex support, distributions can become highly concentrated; this is one reason to inspect simulation output directly.
\end{itemize}
\end{frame}

\begin{frame}{Why use geometrically weighted statistics?}
\begin{itemize}
\item Raw $k$-star or raw shared-partner counts can increase rapidly as a graph becomes more concentrated.
\item A \term{geometrically weighted statistic} gives successive configurations diminishing incremental weight.
\item A \term{decay parameter} controls the rate at which the increment approaches its limiting contribution.
\item \term{Fixed decay} produces a stated canonical ERGM term; \term{estimated decay} produces a curved parameterization.
\item The decay convention is software-specific; every report must state it rather than compare unlabelled numerical values.
\end{itemize}
\end{frame}

\begin{frame}{Geometrically weighted degree}
Let $D_k(y)$ denote the number of actors with degree $k$ in an undirected graph. Under one common convention,
\[
\operatorname{GWD}(y;\alpha)=e^{\alpha}\sum_{k=1}^{n-1}\left\{1-(1-e^{-\alpha})^k\right\}D_k(y), \qquad \alpha>0.
\]
\begin{itemize}
\item The statistic summarizes the \term{entire degree distribution} with decreasing marginal weight for increasingly large degrees.
\item It can represent degree heterogeneity without fitting a separate free coefficient for every possible degree.
\item Directed networks require separate \term{in-degree} and \term{out-degree} considerations; a single undirected degree statistic is not a substitute.
\end{itemize}
\end{frame}

\begin{frame}{Geometrically weighted edgewise shared partners}
Let $EP_k(y)$ be the number of existing edges whose endpoints have exactly $k$ shared partners.
\[
\operatorname{GWESP}(y;\alpha)=e^{\alpha}\sum_{k=1}^{n-2}\left\{1-(1-e^{-\alpha})^k\right\}EP_k(y).
\]
\begin{itemize}
\item \term{Edgewise shared partners} condition on an existing tie and summarize local closure around that tie.
\item In an undirected graph, a positive conditional effect can represent closure under the specified decay; it does not merely mean ``more triangles.''
\item In directed networks, shared-partner and closure orientations differ; transitive and cyclic patterns must be stated rather than called generic clustering.
\end{itemize}
\end{frame}

\begin{frame}{Geometrically weighted dyadwise shared partners}
Let $DP_k(y)$ be the number of dyads whose endpoints have exactly $k$ shared partners.
\[
\operatorname{GWDSP}(y;\alpha)=e^{\alpha}\sum_{k=1}^{n-2}\left\{1-(1-e^{-\alpha})^k\right\}DP_k(y).
\]
\begin{itemize}
\item \term{Dyadwise shared partners} refer to all dyads, not only currently tied pairs.
\item It therefore differs from GWESP both substantively and combinatorially.
\item In Session 1.2, edgewise and dyadwise shared-partner distributions are important \term{GOF diagnostics} for undirected one-mode graphs even when neither is included as a fitted term.
\item Ordinary unipartite shared-partner diagnostics are not automatically meaningful for directed or original bipartite incidence supports.
\end{itemize}
\end{frame}

\begin{frame}{Fixed decay versus estimated decay}
\begin{tabularx}{\textwidth}{>{\bfseries}lX X}
\toprule
Choice & Statistical status & Practical implication\\
\midrule
Fixed $\alpha$ & Canonical ERGM term with a pre-specified decay & Interpret the coefficient conditional on that stated weighting; examine a sensitivity grid if scientifically needed.\\
Estimated $\alpha$ & Curved ERGM parameterization & Inspect projected-score convergence, Monte Carlo uncertainty, repeated fits, and whether the extra flexibility is identified.\\
\bottomrule
\end{tabularx}
\vspace{1ex}
\begin{alertblock}{Discipline}
Estimating a decay because software permits it is not a justification. The scientific configuration, support, and diagnostic evidence must support the added parameter.
\end{alertblock}
\end{frame}

\begin{frame}{Degree heterogeneity is not a nuisance}
\begin{itemize}
\item Degree heterogeneity can reflect activity, popularity, institutional capacity, opportunity, or unmeasured processes.
\item Raw degree counts may be nearly collinear with edges or other terms and can favor a narrow, unstable region of the graph space.
\item A geometrically weighted degree term is a lower-dimensional representation, not evidence that one social mechanism explains all degree variation.
\item \term{GOF obligation:} compare the full simulated degree distribution with the observed distribution; a coefficient alone does not demonstrate reproduction of its shape.
\end{itemize}
\end{frame}

\begin{frame}{Closure is conditional and support-specific}
\begin{itemize}
\item More edges create more possible two-paths; high-degree actors create more opportunities for closure.
\item A closure statistic must be interpreted \term{conditional on density and degree structure} already represented in the model.
\item In directed data, distinguish transitive closure from cyclic closure and identify the oriented relation being modeled.
\item In a bipartite incidence graph, triangles are structurally impossible; a one-mode projection is a different response network, not an automatic diagnostic transformation.
\end{itemize}
\end{frame}

\begin{frame}{A theory-led refinement sequence}
\begin{enumerate}
\item Define the relation, actor boundary, admissible dyads, directionality, and modes.
\item Fit a \term{small baseline} such as density, then add one structural hypothesis at a time.
\item Use a fixed-decay geometrically weighted term before casually estimating multiple decay parameters.
\item Retain models only when MCMC behavior, convergence evidence, simulations, and interpretation remain defensible.
\item Record rejected or unstable refinements; absence of a final complex formula is informative rather than a failure of analysis.
\end{enumerate}
\end{frame}

\begin{frame}{Why raw edges-plus-triangle models can be risky}
\begin{itemize}
\item A positive triangle coefficient rewards every closed triple; the number of opportunities can grow rapidly in denser configurations.
\item Density and closure can reinforce one another, creating sensitivity between sparse and dense graph regimes.
\item A curved closure representation can regularize increments, but does not guarantee a well-behaved fitted distribution.
\item The correct response to instability is diagnosis and transparent simplification, not merely increasing the number of MCMC iterations.
\end{itemize}
\end{frame}

\begin{frame}{What is ERGM degeneracy?}
\begin{itemize}
\item \term{Degeneracy} concerns the shape of a fitted or candidate graph distribution, not merely an estimation warning.
\item In a practically degenerate model, probability may concentrate on a small collection of structurally extreme graphs, such as nearly empty, nearly complete, or otherwise unrealistic regimes.
\item The distribution can be highly sensitive: small parameter changes produce very large changes in expected network statistics.
\item Degeneracy is assessed through the simulated distribution, multimodality, concentration, and the substantive plausibility of simulated graphs.
\end{itemize}
\end{frame}

\begin{frame}{Degeneracy is not a synonym for failure}
\begin{tabularx}{\textwidth}{>{\bfseries}lX}
\toprule
Phenomenon & Meaning\\
\midrule
Poor MCMC mixing & The sampler explores its target distribution inefficiently or insufficiently.\\
MCMLE nonconvergence & \term{Monte Carlo maximum likelihood estimation} has not established a stable estimate with the available Monte Carlo information.\\
Finite-MLE nonexistence & The observed statistic lies on a boundary where an ordinary finite canonical estimate does not exist.\\
Degeneracy & The model distribution itself is concentrated on structurally extreme or separated graph regimes.\\
\bottomrule
\end{tabularx}
\vspace{1ex}
\term{These phenomena can interact, but they are mathematically distinct and require different responses.}
\end{frame}

\begin{frame}{Finite-MLE existence and convex support}
Let
\[
\mathcal{C}=\operatorname{conv}\{g(y):y\in\mathcal{Y}\}
\]
denote the convex support of the sufficient statistics.
\begin{itemize}
\item For a minimal full-rank canonical exponential family with finite support, a finite MLE exists if and only if $g(y_{\mathrm{obs}})\in\operatorname{relint}(\mathcal{C})$, the \term{relative interior}: the interior measured within the affine hull of $\mathcal{C}$.
\item A boundary observation can lead the ordinary MLE toward an extended-family limit rather than a finite coefficient vector.
\item Curved and nonminimal ERGMs require additional qualifications because the parameterization is restricted or redundant.
\item \alert{Boundary behavior is not automatically degeneracy.}
\end{itemize}
\end{frame}

\begin{frame}{Practical symptoms that require investigation}
\begin{itemize}
\item Trace plots show long drifts, abrupt regime changes, or highly persistent autocorrelation.
\item Repeated fits from different starts yield unstable coefficients, decay values, or simulated summaries.
\item Simulated networks collapse repeatedly toward implausibly sparse, dense, disconnected, or highly concentrated regimes.
\item Warnings identify linear dependence, statistics at attainable boundaries, nonidentification, or absent MCMC movement.
\item A GOF envelope systematically misses a substantively important observed feature not forced by the fitted statistics.
\end{itemize}
\end{frame}

\begin{frame}{Convergence for a canonical ERGM}
\begin{itemize}
\item For a finite interior canonical MLE, Monte Carlo estimates of $\E_{\hat\theta}\{g(Y)\}$ should agree with $g(y)$ within numerical and Monte Carlo precision.
\item \term{Trace plots} and autocorrelation assess whether the sampler explores adequately for a fixed parameter value.
\item \term{Repeated estimation} and small estimating-equation discrepancies assess whether the fitting algorithm has stabilized.
\item Neither condition establishes model adequacy: the simulated distribution must still be compared with meaningful observed features.
\end{itemize}
\end{frame}

\begin{frame}{Convergence for a curved ERGM}
\begin{itemize}
\item A curved model is assessed through the projected score
\[
D_{\eta}(\hat\theta)^{\mathsf T}\left[g(y)-\E_{\hat\theta}\{g(Y)\}\right],
\]
relative to its Monte Carlo uncertainty.
\item It is generally incorrect to demand exact componentwise agreement of every count used internally by a curved statistic.
\item Standard errors and information matrices must be interpreted with reference to the parameterization and estimation procedure actually used.
\item If free-decay estimation is weakly identified, a documented fixed-decay sensitivity model may be more transparent than a fragile curved estimate.
\end{itemize}
\end{frame}

\begin{frame}{MCMC trace plots: first diagnostic panel}
\begin{itemize}
\item A \term{trace plot} displays retained MCMC values of monitored sufficient statistics or parameters against sample index.
\item Look for sustained movement around a stable region rather than monotone drift, rare jumps between modes, or very long plateaus.
\item A visually calm trace can still be highly autocorrelated; trace plots and autocorrelation must be read together.
\item Session 1.2 preserves the random seed, burn-in, interval, retained sample size, and messages so a diagnostic run is reproducible.
\end{itemize}
\end{frame}

\begin{frame}{Autocorrelation and effective exploration}
\begin{itemize}
\item The lag-$h$ \term{autocorrelation} measures dependence between retained MCMC values separated by $h$ saved iterations.
\item Slow decay indicates that many saved draws carry similar information; increasing the nominal sample count may not proportionally improve Monte Carlo precision.
\item Thinning changes how often values are retained but does not by itself solve multimodality, poor proposals, or an ill-specified target distribution.
\item High autocorrelation is evidence to investigate computation and specification, not a license to ignore diagnostics because the coefficient table was produced.
\end{itemize}
\end{frame}

\begin{frame}{Sampled-statistic distributions}
\begin{itemize}
\item Histograms or density summaries of retained MCMC statistics reveal concentration, skewness, separated modes, and limited exploration.
\item Compare the observed target statistic with simulated behavior when that comparison is defined by the estimation method.
\item A distribution concentrated far from relevant observed structure can signal a poor fit, a nonconverged algorithm, or a problematic specification; the diagnosis requires the other panels.
\item These displays are \term{diagnostics of computation and model behavior}, not ordinary independent-sample histograms.
\end{itemize}
\end{frame}

\begin{frame}{Simulation-based goodness of fit}
\begin{itemize}
\item Draw $Y^{(1)},\ldots,Y^{(M)}$ from the fitted ERGM and choose an \term{auxiliary diagnostic} $T(Y)$ that is substantively relevant.
\item Compare $T(y_{\mathrm{obs}})$ with the simulated reference distribution $\{T(Y^{(m)})\}_{m=1}^M$.
\item Graphical envelopes, empirical quantiles, and systematic discrepancies are often more informative than a single scalar fit number.
\item The objective is compatibility of selected network features with the fitted graph distribution, not exact reproduction of every realized count.
\end{itemize}
\end{frame}

\begin{frame}{Standard GOF for undirected one-mode networks}
\begin{itemize}
\item \term{Degree distribution:} whether simulated concentration and isolates resemble the observed graph.
\item \term{Geodesic-distance distribution:} whether connected-pair distances and unreachable pairs are represented under a stated convention.
\item \term{Edgewise shared partners:} closure around existing ties.
\item \term{Dyadwise shared partners:} shared-neighbor structure across all dyads.
\item \term{Additional omitted checks:} component sizes, triangle count, and any theory-relevant feature omitted from the formula.
\end{itemize}
\end{frame}

\begin{frame}{Standard GOF for directed one-mode networks}
\begin{itemize}
\item Use separate \term{in-degree} and \term{out-degree} distributions; do not collapse sender activity and receiver popularity into one undirected degree panel.
\item Examine directed geodesic or reachability summaries, declaring how unreachable ordered pairs are handled.
\item A \term{directed triad census} can diagnose transitive, cyclic, and other oriented local configurations when network size and substantive question make it meaningful.
\item Inspect reciprocity, weak or strong component structure, and directed attribute mixing when they are substantively invoked but not directly fitted.
\end{itemize}
\end{frame}

\begin{frame}{Standard GOF for bipartite incidence networks}
\begin{itemize}
\item Preserve the two-mode support and report \term{first-mode} and \term{second-mode} degree distributions separately.
\item A geodesic-distance diagnostic must retain bipartite parity and must not be described as a one-mode social distance without a separately defined projection.
\item Ordinary unipartite triangles, GWESP, ESP, and DSP do not apply directly: within-mode cells are structural zeros and bipartite triangles are impossible.
\item Useful auxiliary checks include component sizes, isolates, woman--woman shared-event overlap, and event--event shared-participant overlap.
\end{itemize}
\end{frame}

\begin{frame}{Model-specific checks are not decorative}
\begin{itemize}
\item \term{Mixing matrices} or assortativity summaries are relevant only when a specific attribute and sender/receiver role have a substantive interpretation.
\item \term{Component-size distributions} are important when connectivity is central to the research question or visibly misrepresented by simulations.
\item \term{Triad census} is a directed structural diagnostic, not a generic replacement for all closure questions.
\item An omitted statistic is most informative when it addresses a scientifically important feature that the model did not force to match.
\end{itemize}
\end{frame}

\begin{frame}{Why inspect omitted statistics?}
\begin{itemize}
\item Agreement on a statistic explicitly fitted by the model is necessary but not strong evidence of broader adequacy.
\item For example, a model with degree and closure terms may still miss component structure, inequality of popularity, directed reachability, or attribute mixing.
\item Select omitted diagnostics from the research question and data-generating design, not by searching until a favorable plot appears.
\item A discrepancy should prompt substantive and statistical review; it does not mechanically dictate another term.
\end{itemize}
\end{frame}

\begin{frame}{GOF is not predictive validation}
\begin{itemize}
\item Structural GOF asks whether networks simulated from the fitted graph distribution reproduce selected observed features.
\item \term{Predictive validation} asks a different question: how well a procedure predicts held-out ties, future networks, or another prespecified target.
\item Sparse-network classification accuracy can be high by predicting absent ties, while structural dependence is badly represented.
\item \term{Causal validation} requires still different assumptions about interventions, confounding, timing, and design.
\end{itemize}
\end{frame}

\begin{frame}{Nested model comparison}
For nested maximum-likelihood ERGMs, the likelihood-ratio statistic is
\[
2\{\ell(\hat\theta_1)-\ell(\hat\theta_0)\}.
\]
\begin{itemize}
\item Under usual interior regularity conditions, a nested comparison has an asymptotic chi-squared reference with degrees of freedom equal to the difference in fitted parameter dimensions.
\item For a curved model, use the difference in $\dim(\theta)$, not the dimension of the expanded natural-parameter representation.
\item Boundary, nonidentification, and other nonregular conditions can invalidate the ordinary reference distribution.
\item A likelihood comparison cannot compensate for poor MCMC behavior, inadequate GOF, or an incoherent scientific specification.
\end{itemize}
\end{frame}

\begin{frame}{Information criteria and their limits}
\begin{itemize}
\item Common summaries are $\operatorname{AIC}=-2\ell(\hat\theta)+2p$ and $\operatorname{BIC}=-2\ell(\hat\theta)+p\log N$.
\item In dependent network data, the effective sample size $N$ is not automatically the number of dyads; its interpretation depends on the asymptotic regime and design.
\item Likelihood calculations may themselves involve Monte Carlo estimation, so small differences can be unstable relative to computational uncertainty.
\item Information criteria are evidence to combine with diagnostics and scientific interpretation, not an automatic model-selection rule.
\end{itemize}
\end{frame}

\begin{frame}{Replication and asymptotic caution}
\begin{itemize}
\item An observed graph is one realization of a graph-valued distribution, not automatically a collection of independent dyads.
\item Relevant asymptotic regimes may involve actors, admissible dyads under restricted dependence, repeated networks, or another explicitly stated replication process.
\item \term{Projectivity} asks whether the model's marginal distribution on an actor subset remains in the same family with parameters consistent with the full-network model; it and consistency under network growth are substantive mathematical properties, not guarantees of an ERGM label.
\item Reports should identify, at least conceptually, the replication or growth regime under which an approximation or generalization is intended.
\end{itemize}
\end{frame}

\begin{frame}{Full Session 1.2 audit workflow}
\begin{enumerate}
\item Validate binary static support, actor boundary, structural zeros, directionality, and modes.
\item State a baseline and one theory-led fixed-decay or curved refinement.
\item Fit with recorded seed, burn-in, interval, retained sample size, and MCMLE settings.
\item Inspect trace, autocorrelation, sampled-statistic distribution, warnings, and repeated-fit stability.
\item Simulate networks for full support-appropriate GOF and omitted-statistic comparisons.
\item Report what was unavailable, unstable, unidentified, or scientifically unsupported; do not conceal an unsuccessful refinement.
\end{enumerate}
\end{frame}

\begin{frame}{Worked example: Florentine business ties}
\begin{itemize}
\item \term{Data:} 16 declared families, including isolates, and an undirected binary business-tie encoding.
\item \term{Stable candidate:} edges plus a fixed-decay geometrically weighted degree term.
\item \term{Curved sensitivity:} estimate the single degree-decay parameter only with projected-score, MCMC, and repeated-fit scrutiny.
\item \term{Standard audit:} degree, geodesic distance, edgewise and dyadwise shared partners; isolate count, components, and triangles as omitted checks.
\item \term{Limit:} a small sparse network does not support confident automatic elaboration of closure terms.
\end{itemize}
\end{frame}

\begin{frame}{Worked example: Sampson liking nominations}
\begin{itemize}
\item \term{Data:} a directed, binary, third-wave liking network among 18 monk-novitiates.
\item \term{Stable candidate:} edges, reciprocity, and a fixed-decay geometrically weighted in-degree term.
\item \term{Design caution:} the observed near-fixed choice pattern makes a curved out-degree specification inappropriate in this guided workflow.
\item \term{Standard audit:} in-degree, out-degree, directed distances, triad census, reciprocity, components, and isolates.
\item \term{Limit:} one cross-sectional nomination wave cannot identify temporal change or causal influence.
\end{itemize}
\end{frame}

\begin{frame}{Worked example: Lazega advice network}
\begin{itemize}
\item \term{Data:} 71 lawyers and directed binary advice nominations; advice is the response layer.
\item \term{Support-valid baseline:} edges and reciprocity, followed by the full directed diagnostic audit.
\item \term{Reproducible pilot result:} fixed-decay in-degree and out-degree refinements produced unconstrained MCMC nonmixing under the documented default run; the app does not present an unstable fit as a worked answer.
\item \term{Standard audit:} separate in/out degree, directed distance, triads, reciprocity, components, isolates, and theory-led attribute mixing.
\item \term{Limit:} an apparently stable baseline does not make popularity/activity unnecessary; other relation layers and attributes are not automatic causal controls.
\end{itemize}
\end{frame}

\begin{frame}{Worked example: Davis Southern Women}
\begin{itemize}
\item \term{Data:} 18 women and 14 events in a binary bipartite attendance-incidence network.
\item \term{Support-aware baseline:} edges plus a first-mode structural degree statistic; the original two-mode support is preserved.
\item \term{Fit boundary lesson:} a proposed bipartite curved refinement can show boundary or linear-dependence problems; the app reports this rather than forcing an invalid model.
\item \term{Standard audit:} mode-specific degree distributions, bipartite distances, components, isolates, and mode-specific overlap summaries.
\item \term{Limit:} one-mode projection and unipartite closure diagnostics concern a different response network.
\end{itemize}
\end{frame}

\begin{frame}{Worked example: Kapferer tailor shop}
\begin{itemize}
\item \term{Data:} 39 workers in an undirected binary first-period sociational relation.
\item \term{Support-valid candidate:} edges plus a fixed-decay edgewise shared-partner closure term.
\item \term{Reproducible pilot result:} the fixed-decay degree-plus-closure refinement produced unconstrained MCMC nonmixing; the app retains the closure candidate and documents the failed joint refinement.
\item \term{Standard audit:} degree, distances, edgewise/dyadwise shared partners, triangles, components, and isolates.
\item \term{Limit:} density, degree concentration, and closure can be confounded; the selected panel and thresholded relation do not represent all social contact or a temporal sequence.
\end{itemize}
\end{frame}

\begin{frame}{BYOD: Session 1.2 upload contract}
\begin{itemize}
\item \term{Node table:} a unique \texttt{id} for each actor and a \texttt{mode} column for a bipartite graph.
\item \term{Edge table:} \texttt{source} and \texttt{target} for binary ties; no self-ties or duplicate dyads after accounting for directionality.
\item \term{Declare explicitly:} directedness, mode structure, actor boundary, structural zeros, observation window, and treatment of missing ties.
\item \term{Starter models:} the app proposes only support-valid fixed-decay or baseline candidates; it rejects valued, temporal, multiplex, and silently projected data.
\item \term{Principle:} an upload enables computation, not automatic appropriateness of a static ERGM.
\end{itemize}
\end{frame}

\begin{frame}{A reporting checklist for Session 1.2}
\begin{itemize}
\item \term{Data and support:} relation, actors, boundary, modes, directionality, structural zeros, missingness, and time window.
\item \term{Specification:} baseline, curved or fixed-decay terms, decay convention, rationale, and any rejected refinement.
\item \term{Estimation:} software, package version, seed, burn-in, interval, retained MCMC sample, iteration controls, and messages.
\item \term{Diagnostics:} trace, autocorrelation, sampled distributions, GOF envelopes, omitted-statistic checks, and repeated-fit evidence.
\item \term{Inference boundary:} conditional graph-distribution interpretation, uncertainty, generalization regime, and no unsupported causal claim.
\end{itemize}
\end{frame}

\begin{frame}{Session 1.2 synthesis}
\begin{itemize}
\item Curved ERGMs use a nonlinear lower-dimensional parameterization; geometrically weighted terms offer stable, parsimonious representations of configuration families.
\item Fixed decay and estimated decay make different statistical commitments and require different diagnostics.
\item Degeneracy, finite-MLE failure, MCMC nonmixing, and MCMLE nonconvergence must be distinguished rather than collapsed into one label.
\item A standard ERGM audit combines MCMC diagnostics, full support-appropriate simulation-based GOF, and substantive omitted-statistic checks.
\item Next: \term{temporal ERGMs for network change} in Day 2, where the modeled object is a conditional transition distribution rather than one static graph distribution.
\end{itemize}
\end{frame}
